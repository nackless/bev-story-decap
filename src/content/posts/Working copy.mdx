import { Image } from 'astro:assets';

const { 
  image, 
  alt = "", 
  caption, 
  width = "35%", 
  borderRadius = "12px", 
  gap = "1.5rem" 
} = Astro.props;

// Determine if the image is an imported asset object or a path string/URL
const isImportedImage = typeof image === 'object' && image !== null;

---

<div class="media-text-left" style={`--img-width: ${width}; --gap: ${gap};`}>
  <div class="media-image-wrapper">
    {isImportedImage ? (
      <Image 
        src={image} 
        alt={alt} 
        style={`border-radius: ${borderRadius};`}
        class="media-img"
      />
    ) : (
      <img 
        src={myPhoto} 
        alt={alt} 
        style={`border-radius: ${borderRadius};`}
        class="media-img"
        loading="lazy"
      />
    )}
    {caption && <p class="caption">{caption}</p>}
  </div>
  <div class="media-content">
    <slot />
  </div>
</div>
